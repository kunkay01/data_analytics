from scipy.stats import pearsonr

# Drop missing values for each pair
df_clean = df_cars[['Price', 'Kilometer', 'PS']].dropna()

# 1) Price vs. Kilometer
corr_pk, p_pk = pearsonr(df_clean['Price'], df_clean['Kilometer'])

# 2) Price vs. PS
corr_pps, p_pps = pearsonr(df_clean['Price'], df_clean['PS'])

# 3) Kilometer vs. PS
corr_kps, p_kps = pearsonr(df_clean['Kilometer'], df_clean['PS'])

print("Correlation results:")
print(f"Price vs Kilometer: r = {corr_pk:.4f}, p = {p_pk:.4e}")
print(f"Price vs PS:        r = {corr_pps:.4f}, p = {p_pps:.4e}")
print(f"Kilometer vs PS:    r = {corr_kps:.4f}, p = {p_kps:.4e}")
